#include <WiFi.h>
#include <esp_now.h>
#include <esp_wifi.h>
#include <DHT.h>
#include <Wire.h>
#include <Adafruit_BMP085.h>


// Definición de pines y tipos
#define DHTPIN 4
#define DHTTYPE DHT22
#define LDR_PIN 34


DHT dht(DHTPIN, DHTTYPE);
Adafruit_BMP085 bmp;
bool bmpOK = false;


// Red del gateway: el emisor NO se conecta a ella, solo la busca para
// ponerse en el MISMO canal Wi-Fi (ESP-NOW solo funciona en el mismo canal)
const char* ssidGateway = "AmoIott";


// Dirección MAC del ESP32 Receptor (Gateway)
uint8_t broadcastAddress[] = {0x20, 0x9B, 0xA9, 0x74, 0x82, 0x18};


// Estructura del Payload (DEBE ser igual en el receptor)
typedef struct struct_message {
  char id[12];              // 12: "EST_MET_01" (10 letras) + '\0' caben de sobra
  float tempDHT;
  float humDHT;
  float pressure;
  float tempBMP;
  int light;
  unsigned int sampleNum;
} struct_message;


struct_message myData;
unsigned int sampleCount = 0;
esp_now_peer_info_t peerInfo;


// Callback para verificar el estado del envío
void OnDataSent(const wifi_tx_info_t *info, esp_now_send_status_t status) {
  Serial.print("Estado del ultimo envio: ");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "Exitoso" : "Fallido");
}


// Busca la red del gateway y devuelve su canal (0 si no la encuentra)
int obtenerCanalWiFi(const char* ssid) {
  int n = WiFi.scanNetworks();
  int canal = 0;
  for (int i = 0; i < n; i++) {
    if (WiFi.SSID(i) == ssid) { canal = WiFi.channel(i); break; }
  }
  WiFi.scanDelete();
  return canal;
}


void setup() {
  Serial.begin(115200);
  dht.begin();


  bmpOK = bmp.begin();
  if (!bmpOK) {
    // Antes: while(1); detenía TODO. Ahora sigue enviando DHT22 y LDR.
    Serial.println("No se pudo iniciar el sensor BMP180. ¡Revisa las conexiones I2C!");
  }


  // Configurar ESP32 en modo Station
  WiFi.mode(WIFI_STA);


  // Ponerse en el canal del router del gateway
  int canal = obtenerCanalWiFi(ssidGateway);
  if (canal > 0) {
    esp_wifi_set_promiscuous(true);
    esp_wifi_set_channel(canal, WIFI_SECOND_CHAN_NONE);
    esp_wifi_set_promiscuous(false);
    Serial.printf("Red %s encontrada en canal %d\n", ssidGateway, canal);
  } else {
    Serial.println("No encontre la red del gateway; uso el canal por defecto (1)");
  }


  if (esp_now_init() != ESP_OK) {
    Serial.println("Error al inicializar ESP-NOW");
    return;
  }
  esp_now_register_send_cb(OnDataSent);


  // Registrar peer (receptor)
  memcpy(peerInfo.peer_addr, broadcastAddress, 6);
  peerInfo.channel = 0;      // 0 = canal actual (el que pusimos arriba)
  peerInfo.encrypt = false;
  if (esp_now_add_peer(&peerInfo) != ESP_OK) {
    Serial.println("Fallo al agregar el peer receptor");
    return;
  }
}


void loop() {
  sampleCount++;


  // Lecturas de sensores
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  float p = bmpOK ? bmp.readPressure() / 100.0F : NAN;   // Convertir a hPa
  float t_bmp = bmpOK ? bmp.readTemperature() : NAN;


  // LDR: promedio de 20 lecturas para quitar ruido
  long suma = 0;
  for (int i = 0; i < 20; i++) { suma += analogRead(LDR_PIN); delay(2); }
  int ldrValue = suma / 20;
  int lightPercent = map(ldrValue, 0, 4095, 0, 100);  // Normalizar a porcentaje (%)
  lightPercent = constrain(lightPercent, 0, 100);


  // Empaquetar los datos
  strncpy(myData.id, "EST_MET_01", sizeof(myData.id));
  myData.tempDHT = t;
  myData.humDHT = h;
  myData.pressure = p;
  myData.tempBMP = t_bmp;
  myData.light = lightPercent;
  myData.sampleNum = sampleCount;


  Serial.printf("\nMuestra %u -> T=%.1f C  H=%.1f %%  P=%.1f hPa  Tbmp=%.1f C  Luz=%d %%\n",
                sampleCount, t, h, p, t_bmp, lightPercent);


  // Enviar mensaje vía ESP-NOW
  esp_err_t result = esp_now_send(broadcastAddress, (uint8_t *) &myData, sizeof(myData));
  if (result == ESP_OK) {
    Serial.println("Paquete enviado correctamente por ESP-NOW");
  } else {
    Serial.println("Error en el envío del paquete");
  }


  delay(20000); // Enviar cada 20 segundos
}
